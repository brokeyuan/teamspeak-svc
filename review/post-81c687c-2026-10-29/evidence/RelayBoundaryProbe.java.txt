package com.tsvoice;
import com.tsvoice.protocol.relay.TsRelayWire;
import com.tsvoice.server.relay.TsRelayRuntime;
import java.net.*;import java.util.*;import java.util.concurrent.*;
public class RelayBoundaryProbe {
 static class Echo implements AutoCloseable {final DatagramSocket socket=new DatagramSocket(0,InetAddress.getLoopbackAddress()); final Thread worker; Echo()throws Exception {worker=new Thread(()->{byte[] b=new byte[8192];while(!socket.isClosed()){try{DatagramPacket p=new DatagramPacket(b,b.length);socket.receive(p);socket.send(new DatagramPacket(p.getData(),p.getLength(),p.getSocketAddress()));}catch(Exception e){if(socket.isClosed())return;}}});worker.setDaemon(true);worker.start();}public void close(){socket.close();}}
 static TsRelayRuntime.RelayConfig config(int port,int pps){return new TsRelayRuntime.RelayConfig(true,Map.of("ts",new TsRelayRuntime.RelayConfig.RelayTargetConfig("127.0.0.1",port)),64,new TsRelayRuntime.Limits(4096,pps,512*1024));}
 static byte[] open(int id){return TsRelayWire.encodeOpen(new TsRelayWire.OpenFrame(id,"ts",TsRelayWire.Purpose.PLAYER_VOICE));}
 static byte[] data(byte[]id,long epoch,byte[]b){return TsRelayWire.encodeData(TsRelayWire.MessageType.DATA_C2S,new TsRelayWire.DataFrame(id,epoch,1,b));}
 static void observed(boolean b,String s){if(!b)throw new AssertionError("Not reproduced: "+s);System.out.println("REPRODUCED: "+s);}
 public static void main(String[]args)throws Exception{
  try(Echo echo=new Echo()){
   CountDownLatch callback=new CountDownLatch(1);boolean[] decodeRejected={false};
   {TsRelayRuntime r=new TsRelayRuntime((id,p,b)->{decodeRejected[0]=TsRelayWire.decode(b)==null;callback.countDown();return !decodeRejected[0];},config(echo.socket.getLocalPort(),256));try{var o=r.onOpen("conn-a","player-a",open(1));r.onData("conn-a",data(o.relaySessionId(),1,new byte[]{1,2,3,4}));observed(callback.await(2,TimeUnit.SECONDS)&&decodeRejected[0],"runtime sends raw UDP bytes to sink; production sink's relay decoder rejects them");}finally{r.close();}}
   BlockingQueue<Integer> sizes=new LinkedBlockingQueue<>();
   {TsRelayRuntime r=new TsRelayRuntime((id,p,b)->{sizes.offer(b.length);return true;},config(echo.socket.getLocalPort(),1));try{var o=r.onOpen("conn-b","player-b",open(2));byte[] sid=o.relaySessionId();r.onRouteBarrier("conn-b",TsRelayWire.encodeRouteBarrier(new TsRelayWire.RouteBarrierFrame(sid,100)));observed(r.onData("conn-b",data(sid,99,new byte[]{7})),"DATA below accepted route barrier is still forwarded");int accepted=0;for(int i=0;i<4;i++)if(r.onData("conn-b",data(sid,100,new byte[]{8})))accepted++;observed(accepted==4,"negotiated maxDatagramsPerSecond=1 does not reject four immediate DATA packets");}finally{r.close();}}
   BlockingQueue<Integer> large=new LinkedBlockingQueue<>();
   {TsRelayRuntime r=new TsRelayRuntime((id,p,b)->{large.offer(b.length);return true;},config(echo.socket.getLocalPort(),256));try{var o=r.onOpen("conn-c","player-c",open(3));observed(r.onData("conn-c",data(o.relaySessionId(),1,new byte[3000])),"3000-byte datagram accepted under 4096 contract");Integer got=large.poll(2,TimeUnit.SECONDS);observed(got!=null&&got==2048,"3000-byte UDP reply silently truncated to 2048 bytes");}finally{r.close();}}
   {TsRelayRuntime r=new TsRelayRuntime((id,p,b)->true,config(echo.socket.getLocalPort(),256));try{var a=r.onOpen("conn-d","player-d",open(1));var b=r.onOpen("conn-e","player-e",open(1));observed(a.ok()&&!b.ok(),"requestId=1 from different MC connections collides in global pendingOpens");}finally{r.close();}}
  }
 }
}
